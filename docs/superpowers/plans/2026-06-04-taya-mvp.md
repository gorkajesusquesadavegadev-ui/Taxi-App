# TAYA MVP — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a thin-vertical taxi MVP: client requests a taxi, nearby free taxistas get it by broadcast, first accept wins, client tracks the taxi live, taxista completes the trip, fare is computed from real distance.

**Architecture:** Spring Boot backend (REST + WebSocket) over PostgreSQL (persistent) and Redis (live location + GEORADIUS). Two lightweight web clients (cliente, taxista) built with Vite + vanilla JS + OpenLayers. Routing/distance via public OSRM with a Haversine fallback. Auth is email+password+JWT.

**Tech Stack:** Java 17, Spring Boot 3.2, Spring Security + jjwt, Spring Data JPA, Spring Data Redis, Spring WebSocket, PostgreSQL 16, Redis 7.2, Maven, JUnit 5 + Mockito + Testcontainers; Vite + OpenLayers (npm `ol`) for clients; Docker Compose for infra.

**Reference spec:** `docs/superpowers/specs/2026-06-04-taya-mvp-design.md`

```java
// security/JwtTokenProvider.java
package com.taya.security;
import io.jsonwebtoken.*;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.util.Date;

@Component
public class JwtTokenProvider {
    private final SecretKey key;
    private final long expirationMs;

    public JwtTokenProvider(@Value("${jwt.secret}") String secret,
                            @Value("${jwt.expiration}") long expirationMs) {
        this.key = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
        this.expirationMs = expirationMs;
    }

    public String generate(Long userId, String rol) {
        Date now = new Date();
        return Jwts.builder()
            .setSubject(String.valueOf(userId))
            .claim("rol", rol)
            .setIssuedAt(now)
            .setExpiration(new Date(now.getTime() + expirationMs))
            .signWith(key, SignatureAlgorithm.HS256)
            .compact();
    }

    private Claims parse(String token) {
        return Jwts.parserBuilder().setSigningKey(key).build()
            .parseClaimsJws(token).getBody();
    }
    public boolean isValid(String token) {
        try { parse(token); return true; } catch (JwtException | IllegalArgumentException e) { return false; }
    }
    public Long getUserId(String token) { return Long.valueOf(parse(token).getSubject()); }
    public String getRol(String token) { return parse(token).get("rol", String.class); }
}
```

Note: HS256 needs a ≥32-byte secret; the dev value above is 40 chars. Run: `./mvnw -q -Dtest=JwtTokenProviderTest test` → Expected: PASS.

```java
package com.taya.security;
import com.taya.repository.UsuarioRepository;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.userdetails.*;
import org.springframework.stereotype.Service;
import java.util.List;

@Service
public class CustomUserDetailsService implements UserDetailsService {
    private final UsuarioRepository usuarios;
    public CustomUserDetailsService(UsuarioRepository usuarios) { this.usuarios = usuarios; }

    @Override
    public UserDetails loadUserByUsername(String userId) throws UsernameNotFoundException {
        var u = usuarios.findById(Long.valueOf(userId))
            .orElseThrow(() -> new UsernameNotFoundException(userId));
        return new User(String.valueOf(u.getId()), u.getPasswordHash(),
            List.of(new SimpleGrantedAuthority("ROLE_" + u.getRol().name())));
    }
}
```

```java
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.*;
import org.springframework.lang.NonNull;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;
import java.io.IOException;
import java.util.List;

@Component
public class JwtAuthFilter extends OncePerRequestFilter {
    private final JwtTokenProvider jwt;
    public JwtAuthFilter(JwtTokenProvider jwt) { this.jwt = jwt; }

    @Override
    protected void doFilterInternal(@NonNull HttpServletRequest req, @NonNull HttpServletResponse res,
                                    @NonNull FilterChain chain) throws ServletException, IOException {
        String header = req.getHeader("Authorization");
        if (header != null && header.startsWith("Bearer ")) {
            String token = header.substring(7);
            if (jwt.isValid(token)) {
                Long userId = jwt.getUserId(token);
                var auth = new UsernamePasswordAuthenticationToken(
                    String.valueOf(userId), null,
                    List.of(new SimpleGrantedAuthority("ROLE_" + jwt.getRol(token))));
                SecurityContextHolder.getContext().setAuthentication(auth);
            }
        }
        chain.doFilter(req, res);
    }
}
```

```java
// util/GeoUtil.java
package com.taya.util;
public final class GeoUtil {
    private GeoUtil() {}
    private static final double R = 6371.0; // km
    public static double haversineKm(double lat1, double lng1, double lat2, double lng2) {
        double dLat = Math.toRadians(lat2 - lat1), dLng = Math.toRadians(lng2 - lng1);
        double a = Math.sin(dLat/2)*Math.sin(dLat/2)
            + Math.cos(Math.toRadians(lat1))*Math.cos(Math.toRadians(lat2))
            *Math.sin(dLng/2)*Math.sin(dLng/2);
        return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    }
}
```

```java
import org.springframework.data.geo.*;

import org.springframework.stereotype.Service;
import java.time.Duration;
import java.time.Instant;
import java.util.ArrayList;
import java.util.List;

@Service
public class UbicacionService {
    private final StringRedisTemplate redis;
    public UbicacionService(StringRedisTemplate redis) { this.redis = redis; }

    private static final String GEO_LIBRES = "taxis:libres";

    public void actualizarUbicacion(Long taxistaId, double lat, double lng) {
        redis.opsForValue().set("taxi:" + taxistaId + ":ubicacion", lat + "," + lng, Duration.ofSeconds(10));
    }

    public void heartbeat(Long taxistaId) {
        redis.opsForValue().set("taxi:" + taxistaId + ":last_ping", Instant.now().toString(), Duration.ofSeconds(30));
    }

    public void setLibre(Long taxistaId, double lat, double lng) {
        redis.opsForValue().set("taxi:" + taxistaId + ":estado", "LIBRE");
        redis.opsForGeo().add(GEO_LIBRES, new Point(lng, lat), String.valueOf(taxistaId));
        heartbeat(taxistaId);
    }

    public void setOcupado(Long taxistaId) {
        redis.opsForValue().set("taxi:" + taxistaId + ":estado", "OCUPADO");
        redis.opsForGeo().remove(GEO_LIBRES, String.valueOf(taxistaId));
    }

    public void setDesconectado(Long taxistaId) {
        redis.opsForValue().set("taxi:" + taxistaId + ":estado", "DESCONECTADO");
        redis.opsForGeo().remove(GEO_LIBRES, String.valueOf(taxistaId));
    }

    public boolean tienePing(Long taxistaId) {
        return Boolean.TRUE.equals(redis.hasKey("taxi:" + taxistaId + ":last_ping"));
    }

    public String ubicacion(Long taxistaId) {
        return redis.opsForValue().get("taxi:" + taxistaId + ":ubicacion");
    }

    /** Free taxistas within radiusKm of (lat,lng). */
    public List<TaxiCercano> libresEnRadio(double lat, double lng, double radiusKm) {
        var circle = new Circle(new Point(lng, lat), new Distance(radiusKm, Metrics.KILOMETERS));
        var args = RedisGeoCommands.GeoRadiusCommandArgs.newGeoRadiusArgs()
            .includeCoordinates().sortAscending();
        var res = redis.opsForGeo().radius(GEO_LIBRES, circle, args);
        List<TaxiCercano> out = new ArrayList<>();
        if (res != null) {
            for (var loc : res.getContent()) {
                Long id = Long.valueOf(loc.getContent().getName());
                Point p = loc.getContent().getPoint();
                out.add(new TaxiCercano(id, p.getY(), p.getX())); // y=lat, x=lng
            }
        }
        return out;
    }

    public record TaxiCercano(Long taxistaId, double lat, double lng) {}
}
```

```java
    private final UbicacionService ubicacion;
    public HeartbeatScheduler(StringRedisTemplate redis, UbicacionService ubicacion) {
        this.redis = redis; this.ubicacion = ubicacion;
    }

    /** Drop free taxistas whose heartbeat expired (no last_ping key). */
    @Scheduled(fixedRate = 15000)
    public void limpiarDesconectados() {
        // Enumerate free taxistas via the parallel id set maintained by UbicacionService.
        Set<String> ids = redis.opsForSet().members("taxis:libres:ids");
        if (ids == null) return;
        for (String id : ids) {
            if (!ubicacion.tienePing(Long.valueOf(id))) {
                ubicacion.setDesconectado(Long.valueOf(id)); // also removes from taxis:libres:ids
            }
        }
    }
}
```

```java
package com.taya.service;
import com.taya.util.GeoUtil;
import org.springframework.stereotype.Service;
import org.springframework.web.client.RestClient;

@Service
public class OsrmClient {
    private final RestClient http = RestClient.create();
    private static final String BASE = "https://router.project-osrm.org/route/v1/driving/";

    /** Driving distance in km; falls back to Haversine on any error. */
    public double distanciaKm(double oLat, double oLng, double dLat, double dLng) {
        try {
            String url = BASE + oLng + "," + oLat + ";" + dLng + "," + dLat + "?overview=false";
            OsrmResponse r = http.get().uri(url).retrieve().body(OsrmResponse.class);
            if (r != null && r.routes() != null && !r.routes().isEmpty())
                return r.routes().get(0).distance() / 1000.0;
        } catch (Exception ignored) { }
        return GeoUtil.haversineKm(oLat, oLng, dLat, dLng);
    }

    record OsrmResponse(java.util.List<Route> routes) {}
    record Route(double distance) {}
}
```

```java
package com.taya.service;
import com.taya.dto.response.EstimacionResponse;
import com.taya.model.ConfiguracionTarifa;
import com.taya.repository.ConfiguracionTarifaRepository;
import org.springframework.stereotype.Service;
import java.math.*;

@Service
public class TarifaService {
    private final ConfiguracionTarifaRepository repo;
    private final OsrmClient osrm;
    public TarifaService(ConfiguracionTarifaRepository repo, OsrmClient osrm) { this.repo = repo; this.osrm = osrm; }

    private ConfiguracionTarifa cfg() {
        return repo.findAll().stream().findFirst()
            .orElseThrow(() -> new IllegalStateException("configuracion_tarifa vacía"));
    }
    private BigDecimal calcular(ConfiguracionTarifa c, double km) {
        return c.getBajadaBandera()
            .add(c.getPrecioKm().multiply(BigDecimal.valueOf(km)))
            .setScale(2, RoundingMode.HALF_UP);
    }

    public EstimacionResponse estimar(double oLat, double oLng, double dLat, double dLng) {
        var c = cfg();
        double km = osrm.distanciaKm(oLat, oLng, dLat, dLng);
        return new EstimacionResponse(BigDecimal.valueOf(km), calcular(c, km));
    }
    public BigDecimal precioFinal(double distanciaRealKm) { return calcular(cfg(), distanciaRealKm); }
    public BigDecimal radioBusquedaKm() { return cfg().getRadioBusquedaKm(); }
}
```

```java
// service/ViajeService.java
package com.taya.service;
import com.taya.dto.request.*;
import com.taya.dto.response.*;
import com.taya.model.*;
import com.taya.repository.ViajeRepository;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;
import java.math.BigDecimal;
import java.time.Instant;
import java.util.List;

@Service
public class ViajeService {
    private final ViajeRepository viajes;
    private final TarifaService tarifa;
    private final UbicacionService ubicacion;
    private final MatchingService matching;

    public ViajeService(ViajeRepository v, TarifaService t, UbicacionService u, MatchingService m) {
        this.viajes = v; this.tarifa = t; this.ubicacion = u; this.matching = m;
    }

    public EstimacionResponse estimar(EstimarRequest r) {
        return tarifa.estimar(r.origenLat(), r.origenLng(), r.destinoLat(), r.destinoLng());
    }

    @Transactional
    public ViajeResponse solicitar(Long clienteId, SolicitarViajeRequest r) {
        var est = tarifa.estimar(r.origenLat(), r.origenLng(), r.destinoLat(), r.destinoLng());
        Viaje v = new Viaje();
        v.setClienteId(clienteId);
        v.setOrigenLat(r.origenLat()); v.setOrigenLng(r.origenLng());
        v.setDestinoLat(r.destinoLat()); v.setDestinoLng(r.destinoLng());
        v.setOrigenTexto(r.origenTexto()); v.setDestinoTexto(r.destinoTexto());
        v.setDistanciaEstimadaKm(est.distanciaKm()); v.setPrecioEstimado(est.precioEstimado());
        v.setEstado(EstadoViaje.SOLICITADO);
        v = viajes.save(v);
        matching.difundirNuevoViaje(v);
        return ViajeResponse.of(v);
    }

    @Transactional
    public ViajeResponse aceptar(Long viajeId, Long taxistaId) {
        int updated = viajes.aceptarSiDisponible(viajeId, taxistaId);
        if (updated == 0)
            throw new ResponseStatusException(HttpStatus.CONFLICT, "viaje ya no disponible");
        ubicacion.setOcupado(taxistaId);
        return ViajeResponse.of(viajes.findById(viajeId).orElseThrow());
    }

    private Viaje cargarDelTaxista(Long viajeId, Long taxistaId, EstadoViaje requerido) {
        Viaje v = viajes.findById(viajeId)
            .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND, "viaje no existe"));
        if (!taxistaId.equals(v.getTaxistaId()))
            throw new ResponseStatusException(HttpStatus.FORBIDDEN, "no es tu viaje");
        if (v.getEstado() != requerido)
            throw new ResponseStatusException(HttpStatus.CONFLICT, "estado inválido: " + v.getEstado());
        return v;
    }

    @Transactional
    public ViajeResponse llegada(Long viajeId, Long taxistaId) {
        Viaje v = cargarDelTaxista(viajeId, taxistaId, EstadoViaje.ACEPTADO);
        v.setEstado(EstadoViaje.LLEGADO); v.setLlegadaEn(Instant.now());
        return ViajeResponse.of(viajes.save(v));
    }

    @Transactional
    public ViajeResponse iniciar(Long viajeId, Long taxistaId) {
        Viaje v = cargarDelTaxista(viajeId, taxistaId, EstadoViaje.LLEGADO);
        v.setEstado(EstadoViaje.EN_CURSO); v.setIniciadoEn(Instant.now());
        return ViajeResponse.of(viajes.save(v));
    }

    @Transactional
    public ViajeResponse finalizar(Long viajeId, Long taxistaId, double distanciaReal) {
        Viaje v = cargarDelTaxista(viajeId, taxistaId, EstadoViaje.EN_CURSO);
        v.setDistanciaRealKm(BigDecimal.valueOf(distanciaReal));
        v.setPrecioFinal(tarifa.precioFinal(distanciaReal));
        v.setEstado(EstadoViaje.COMPLETADO); v.setCompletadoEn(Instant.now());
        ubicacion.setLibre(taxistaId, v.getDestinoLat(), v.getDestinoLng());
        return ViajeResponse.of(viajes.save(v));
    }

    @Transactional
    public ViajeResponse cancelar(Long viajeId, Long usuarioId) {
        Viaje v = viajes.findById(viajeId)
            .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND, "viaje no existe"));
        if (!usuarioId.equals(v.getClienteId()) && !usuarioId.equals(v.getTaxistaId()))
            throw new ResponseStatusException(HttpStatus.FORBIDDEN, "no participas en el viaje");
        if (v.getEstado() == EstadoViaje.COMPLETADO || v.getEstado() == EstadoViaje.CANCELADO)
            throw new ResponseStatusException(HttpStatus.CONFLICT, "viaje ya cerrado");
        if (v.getTaxistaId() != null) ubicacion.setLibre(v.getTaxistaId(),
            v.getOrigenLat(), v.getOrigenLng());
        v.setEstado(EstadoViaje.CANCELADO); v.setCanceladoEn(Instant.now());
        return ViajeResponse.of(viajes.save(v));
    }

    public List<ViajeResponse> historialCliente(Long clienteId) {
        return viajes.findByClienteIdOrderBySolicitadoEnDesc(clienteId).stream().map(ViajeResponse::of).toList();
    }
    public List<ViajeResponse> historialTaxista(Long taxistaId) {
        return viajes.findByTaxistaIdOrderBySolicitadoEnDesc(taxistaId).stream().map(ViajeResponse::of).toList();
    }
}
```

```javascript
const BASE = "http://localhost:8080/api";

export function setToken(t) { localStorage.setItem("taya_token", t); }
export function getToken() { return localStorage.getItem("taya_token"); }
export function clearToken() { localStorage.removeItem("taya_token"); }

async function req(method, path, body) {
  const headers = { "Content-Type": "application/json" };
  const token = getToken();
  if (token) headers["Authorization"] = "Bearer " + token;
  const res = await fetch(BASE + path, {
    method, headers, body: body ? JSON.stringify(body) : undefined,
  });
  if (!res.ok) throw new Error((await res.text()) || res.status);
  const text = await res.text();
  return text ? JSON.parse(text) : null;
}
```

```javascript
export function createMap(targetId) {
  const markers = new VectorSource();
  const map = new Map({
    target: targetId,
    layers: [new TileLayer({ source: new OSM() }), new VectorLayer({ source: markers })],
    view: new View({ center: fromLonLat(LPGC), zoom: 13 }),
  });
  return {
    map, markers,
    clearMarkers: () => markers.clear(),
    addMarker: (lng, lat, color = "#1565c0") => {
      const f = new Feature(new Point(fromLonLat([lng, lat])));
      f.setStyle(new Style({ image: new CircleStyle({ radius: 8, fill: new Fill({ color }) }) }));
      markers.addFeature(f);
      return f;
    },
    moveFeature: (f, lng, lat) => f.getGeometry().setCoordinates(fromLonLat([lng, lat])),
  };
}
```

```java
// websocket/TaxiWsHandler.java
package com.taya.websocket;
import com.fasterxml.jackson.databind.*;
import com.taya.security.JwtTokenProvider;
import com.taya.service.UbicacionService;
import org.springframework.stereotype.Component;
import org.springframework.web.socket.*;
import org.springframework.web.socket.handler.TextWebSocketHandler;
import java.net.URI;
import java.util.Map;

@Component
public class TaxiWsHandler extends TextWebSocketHandler {
    private final JwtTokenProvider jwt;
    private final WsSessionRegistry registry;
    private final UbicacionService ubicacion;
    private final ObjectMapper mapper = new ObjectMapper();
    public TaxiWsHandler(JwtTokenProvider jwt, WsSessionRegistry registry, UbicacionService ubicacion) {
        this.jwt = jwt; this.registry = registry; this.ubicacion = ubicacion;
    }

    private Long userIdOf(WebSocketSession s) { return (Long) s.getAttributes().get("userId"); }

    @Override public void afterConnectionEstablished(WebSocketSession session) {
        String token = tokenFromUri(session.getUri());
        if (token == null || !jwt.isValid(token)) {
            try { session.close(CloseStatus.POLICY_VIOLATION); } catch (Exception ignored) {} return;
        }
        Long userId = jwt.getUserId(token);
        session.getAttributes().put("userId", userId);
        session.getAttributes().put("rol", jwt.getRol(token));
        registry.register(userId, session);
    }

    @Override protected void handleTextMessage(WebSocketSession session, TextMessage message) throws Exception {
        Long userId = userIdOf(session);
        JsonNode node = mapper.readTree(message.getPayload());
        String event = node.path("event").asText();
        JsonNode data = node.path("data");
        switch (event) {
            case "location:update" -> ubicacion.actualizarUbicacion(userId, data.get("lat").asDouble(), data.get("lng").asDouble());
            case "heartbeat" -> ubicacion.heartbeat(userId);
            default -> { /* ignore unknown */ }
        }
    }

    @Override public void afterConnectionClosed(WebSocketSession session, CloseStatus status) {
        Long userId = userIdOf(session);
        if (userId != null) registry.remove(userId);
    }

    private String tokenFromUri(URI uri) {
        if (uri == null || uri.getQuery() == null) return null;
        for (String kv : uri.getQuery().split("&")) {
            var p = kv.split("=", 2);
            if (p.length == 2 && p[0].equals("token")) return p[1];
        }
        return null;
    }
}
```

```java
// websocket/TripWsServiceImpl.java
package com.taya.websocket;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.taya.model.Viaje;
import org.springframework.stereotype.Service;
import org.springframework.web.socket.TextMessage;
import org.springframework.web.socket.WebSocketSession;
import java.util.List;
import java.util.Map;

@Service
public class TripWsServiceImpl implements TripWsService {
    private final WsSessionRegistry registry;
    private final ObjectMapper mapper = new ObjectMapper();
    public TripWsServiceImpl(WsSessionRegistry registry) { this.registry = registry; }

    private void send(Long userId, String event, Object data) {
        WebSocketSession s = registry.get(userId);
        if (s == null || !s.isOpen()) return;
        try { s.sendMessage(new TextMessage(mapper.writeValueAsString(Map.of("event", event, "data", data)))); }
        catch (Exception ignored) {}
    }

    @Override public void enviarNuevoViaje(List<Long> taxistaIds, Viaje v) {
        var payload = Map.of("viajeId", v.getId(), "origenLat", v.getOrigenLat(), "origenLng", v.getOrigenLng(),
            "destinoLat", v.getDestinoLat(), "destinoLng", v.getDestinoLng(), "precioEstimado", v.getPrecioEstimado());
        taxistaIds.forEach(id -> send(id, "trip:new", payload));
    }
    @Override public void enviarAceptado(Long clienteId, Viaje v) {
        send(clienteId, "trip:accepted", Map.of("viajeId", v.getId(), "taxistaId", v.getTaxistaId()));
    }
    @Override public void enviarTomado(List<Long> taxistaIds, Long viajeId) {
        taxistaIds.forEach(id -> send(id, "trip:taken", Map.of("viajeId", viajeId)));
    }
    @Override public void enviarUbicacionTaxi(Long clienteId, double lat, double lng) {
        send(clienteId, "trip:location", Map.of("lat", lat, "lng", lng));
    }
    @Override public void enviarLlegada(Long clienteId, Long viajeId) {
        send(clienteId, "trip:arrived", Map.of("viajeId", viajeId));
    }
    @Override public void enviarEstado(Long clienteId, String status) {
        send(clienteId, "trip:status", Map.of("status", status));
    }
    @Override public void enviarCancelado(Long clienteId, Long taxistaId, Long viajeId) {
        if (clienteId != null) send(clienteId, "trip:cancelled", Map.of("viajeId", viajeId));
        if (taxistaId != null) send(taxistaId, "trip:cancelled", Map.of("viajeId", viajeId));
    }
}
```
